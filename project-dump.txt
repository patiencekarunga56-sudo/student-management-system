<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;

class AuthController extends \Illuminate\Routing\Controller
{
    public function showLogin()
    {
        return view('auth.login');
    }

    public function login(Request $request)
    {
        $credentials = $request->validate([
            'email' => 'required|email',
            'password' => 'required'
        ]);

        if (Auth::attempt($credentials)) {
            $request->session()->regenerate();

            $user = Auth::user();

            // Lecturer / Administrator must be approved before entering dashboard
            if (in_array($user->role, ['Lecturer', 'Administrator']) && ! $user->is_approved) {
                return redirect()->route('pending.approval');
            }

            return redirect()->intended('dashboard');
        }

        return back()->withErrors(['email' => 'Invalid credentials'])->onlyInput('email');
    }

    public function showRegister()
    {
        return view('auth.register');
    }

    public function register(Request $request)
    {
        $data = $request->validate([
            'name' => 'required|string|max:255',
            'email' => 'required|string|email|max:255|unique:users',
            'password' => 'required|string|min:8|confirmed',
            'role' => 'required|in:Student,Lecturer,Administrator',
        ]);

        $isStudent = $data['role'] === 'Student';

        $user = User::create([
            'name' => $data['name'],
            'email' => $data['email'],
            'password' => Hash::make($data['password']),
            'role' => $data['role'],
            'is_approved' => $isStudent,
            'approved_at' => $isStudent ? now() : null,
        ]);

        // Assign Spatie role so role:Lecturer / role:Administrator middleware works
        $user->assignRole($data['role']);

        Auth::login($user);
        $request->session()->regenerate();

        // Students go straight to dashboard; Lecturer/Admin wait for approval
        if ($isStudent) {
            return redirect()->route('dashboard');
        }

        return redirect()->route('pending.approval');
    }

    public function logout(Request $request)
    {
        Auth::logout();
        $request->session()->invalidate();
        $request->session()->regenerateToken();
        return redirect('/');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Course;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Routing\Controllers\HasMiddleware;
use Illuminate\Routing\Controllers\Middleware;

class CourseController extends Controller implements HasMiddleware
{
    public static function middleware(): array
    {
        return [
            new Middleware('auth'),
            new Middleware('permission:manage courses', except: ['index', 'show']),
        ];
    }

    public function index()
    {
        $coursesQuery = Course::with('lecturer');

        if (auth()->user()->hasRole('Lecturer')) {
            $coursesQuery->where('lecturer_id', auth()->id());
        }

        $courses = $coursesQuery->get();

        return view('courses.index', compact('courses'));
    }

    public function create()
    {
        $lecturers = User::where('role', 'Lecturer')->get();

        return view('courses.create', compact('lecturers'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'code' => 'required|string|unique:courses',
            'lecturer_id' => 'nullable|exists:users,id',
        ]);

        Course::create($request->all());

        return redirect()->route('courses.index')->with('success', 'Course created successfully.');
    }

    public function show(Course $course)
    {
        $user = auth()->user();
        $isAssignedLecturer = $user->hasRole('Lecturer')
            && (int) $course->lecturer_id === (int) $user->getKey();

        if ($user->hasRole('Lecturer') && ! $isAssignedLecturer) {
            abort(403, 'You can only view rosters for your assigned courses.');
        }

        $canViewRoster = $user->hasRole('Administrator') || $isAssignedLecturer;

        $course->load('lecturer');

        if ($canViewRoster) {
            $course->load('students');
        }

        return view('courses.show', compact('course', 'canViewRoster'));
    }

    public function edit(Course $course)
    {
        $lecturers = User::where('role', 'Lecturer')->get();

        return view('courses.edit', compact('course', 'lecturers'));
    }

    public function update(Request $request, Course $course)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'code' => 'required|string|unique:courses,code,'.$course->id,
            'lecturer_id' => 'nullable|exists:users,id',
        ]);

        $course->update($request->all());

        return redirect()->route('courses.index')->with('success', 'Course updated successfully.');
    }

    public function destroy(Course $course)
    {
        $course->delete();

        return redirect()->route('courses.index')->with('success', 'Course deleted successfully.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Course;
use App\Models\Registration;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Routing\Controllers\HasMiddleware;
use Illuminate\Routing\Controllers\Middleware;
use Illuminate\Support\Facades\Auth;
use Illuminate\Validation\Rule;

class RegistrationController extends Controller implements HasMiddleware
{
    public static function middleware(): array
    {
        return [
            new Middleware('auth'),
        ];
    }

    public function index()
    {
        $canManageRegistrations = Auth::user()->can('manage registrations');

        $registrations = $canManageRegistrations
            ? Registration::with(['user', 'course'])->latest()->get()
            : Auth::user()->registrations()->with('course')->latest()->get();

        return view('registrations.index', compact('registrations', 'canManageRegistrations'));
    }

    public function create()
    {
        $canManageRegistrations = Auth::user()->can('manage registrations');
        $registeredCourseIds = $canManageRegistrations
            ? collect()
            : Auth::user()->registrations()->pluck('course_id');
        $courses = Course::whereNotIn('id', $registeredCourseIds)->orderBy('name')->get();
        $students = $canManageRegistrations
            ? User::where('role', 'Student')->orderBy('name')->get()
            : collect();

        return view('registrations.create', compact('courses', 'students', 'canManageRegistrations'));
    }

    public function store(Request $request)
    {
        $canManageRegistrations = Auth::user()->can('manage registrations');
        $validated = $request->validate([
            'course_id' => ['required', 'exists:courses,id'],
            'user_id' => $canManageRegistrations
                ? ['required', Rule::exists('users', 'id')->where('role', 'Student')]
                : ['prohibited'],
        ]);

        $userId = $canManageRegistrations ? (int) $validated['user_id'] : Auth::id();

        $exists = Registration::where('user_id', $userId)
            ->where('course_id', $request->course_id)
            ->exists();

        if ($exists) {
            return back()->withInput()->with('error', 'This student is already registered for that course.');
        }

        Registration::firstOrCreate([
            'user_id' => $userId,
            'course_id' => $request->course_id,
        ]);

        return redirect()->route('registrations.index')->with('success', 'Course registration saved successfully.');
    }

    public function destroy(Registration $registration)
    {
        if ($registration->user_id !== Auth::id() && ! Auth::user()->can('manage registrations')) {
            abort(403, 'Unauthorized action.');
        }

        $registration->delete();

        return back()->with('success', 'Registration removed.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Routing\Controllers\HasMiddleware;
use Illuminate\Routing\Controllers\Middleware;

class UserController extends Controller implements HasMiddleware
{
    public static function middleware(): array
    {
        return [
            new Middleware('auth'),
            new Middleware('approved'), // <-- Added so unapproved admins can't view this
            new Middleware('role:Administrator'),
        ];
    }

    public function index()
    {
        $users = User::all();

        return view('users.index', compact('users'));
    }

    // Show the list of pending users
    public function pending()
    {
        $users = User::whereIn('role', ['Lecturer', 'Administrator'])
            ->where('is_approved', false)
            ->latest()
            ->get();

        return view('users.pending', compact('users'));
    }

    // Approve the user
    public function approve(User $user)
    {
        $user->update([
            'is_approved' => true,
            'approved_at' => now(),
        ]);

        return back()->with('success', "{$user->name} has been approved!");
    }

    public function update(Request $request, User $user)
    {
        $request->validate([
            'role' => 'required|in:Administrator,Lecturer,Student',
        ]);

        $roleChanged = $user->role !== $request->role;
        $requiresApproval = in_array($request->role, ['Administrator', 'Lecturer']);
        $requiresNewApproval = $roleChanged && $requiresApproval;

        $user->update([
            'role' => $request->role,
            'is_approved' => $requiresNewApproval ? false : ($requiresApproval ? $user->is_approved : true),
            'approved_at' => $requiresNewApproval ? null : $user->approved_at,
        ]);
        $user->syncRoles($request->role);

        return back()->with('success', 'User role updated successfully.');
    }
}

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureUserIsApproved
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = $request->user();

        // If the user is a Lecturer or Administrator and is NOT approved, block them.
        if ($user && in_array($user->role, ['Lecturer', 'Administrator']) && ! $user->is_approved) {
            return redirect()->route('pending.approval');
        }

        return $next($request);
    }
}

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class RoleMiddleware
{
    public function handle(Request $request, Closure $next, string $role): Response
    {
        // If the user is not logged in OR their role doesn't match
        if (! $request->user() || $request->user()->role !== $role) {
            abort(403, 'Unauthorized access. You do not have the correct role.');
        }

        return $next($request);
    }
}
